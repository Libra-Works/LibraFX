cmake_minimum_required(VERSION 3.16)
project(LiberaFxRuntime LANGUAGES CXX)

find_package(Threads REQUIRED)
add_library(liberafx_runtime STATIC Core/VfxRuntime.cpp Core/LivePreview.cpp Core/BuiltinNodes.cpp Core/DelayNode.cpp Core/GeneratorPool.cpp Core/NamedIo.cpp Core/EffectLifecycle.cpp Core/SpawnablePackage.cpp Core/SpawnableHost.cpp)
add_library(LibraRuntimeCore ALIAS liberafx_runtime)
target_include_directories(liberafx_runtime PUBLIC include)
target_compile_features(liberafx_runtime PUBLIC cxx_std_17)
target_link_libraries(liberafx_runtime PUBLIC Threads::Threads)
set_target_properties(liberafx_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)

option(LIBERAFX_BUILD_INTEROP "Build the native C ABI plugin" ON)
if(LIBERAFX_BUILD_INTEROP)
    add_library(liberafx_native SHARED Core/Interop.cpp)
    target_link_libraries(liberafx_native PRIVATE liberafx_runtime Threads::Threads)
    target_include_directories(liberafx_native PUBLIC include)
    target_compile_features(liberafx_native PRIVATE cxx_std_17)
    target_compile_definitions(liberafx_native PRIVATE LIBRA_INTEROP_BUILD=1)
    set_target_properties(liberafx_runtime liberafx_native PROPERTIES
        CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
endif()

option(LIBERAFX_BUILD_TESTS "Build runtime tests" OFF)
option(LIBERAFX_MATH_FORCE_SCALAR "Disable native SIMD math" OFF)
if(LIBERAFX_MATH_FORCE_SCALAR)
    target_compile_definitions(liberafx_runtime PUBLIC LIBERAFX_MATH_FORCE_SCALAR=1)
endif()
if(LIBERAFX_BUILD_TESTS)
    enable_testing()
    add_executable(liberafx_runtime_tests tests/VfxRuntimeTests.cpp)
    target_link_libraries(liberafx_runtime_tests PRIVATE liberafx_runtime Threads::Threads)
    add_test(NAME liberafx_runtime_tests COMMAND liberafx_runtime_tests)
    add_executable(liberafx_generator_pool_tests tests/GeneratorPoolTests.cpp)
    target_link_libraries(liberafx_generator_pool_tests PRIVATE liberafx_runtime Threads::Threads)
    add_test(NAME liberafx_generator_pool_tests COMMAND liberafx_generator_pool_tests)
    add_executable(liberafx_lifetime_tests tests/LifetimeTests.cpp)
    target_link_libraries(liberafx_lifetime_tests PRIVATE liberafx_runtime)
    add_test(NAME liberafx_lifetime_tests COMMAND liberafx_lifetime_tests)
    add_executable(liberafx_effect_lifecycle_tests tests/EffectLifecycleTests.cpp)
    target_link_libraries(liberafx_effect_lifecycle_tests PRIVATE liberafx_runtime)
    add_test(NAME liberafx_effect_lifecycle_tests COMMAND liberafx_effect_lifecycle_tests)
    add_executable(liberafx_solar_system_matrix_tests tests/SolarSystemMatrixTests.cpp)
    target_link_libraries(liberafx_solar_system_matrix_tests PRIVATE liberafx_runtime)

    target_link_libraries(liberafx_render_queue_tests PRIVATE liberafx_runtime Threads::Threads)
    add_test(NAME liberafx_render_queue_tests COMMAND liberafx_render_queue_tests)
    add_executable(liberafx_live_preview_tests tests/LivePreviewTests.cpp)
    target_link_libraries(liberafx_live_preview_tests PRIVATE liberafx_runtime)
    add_test(NAME liberafx_live_preview_tests COMMAND liberafx_live_preview_tests)
    add_executable(liberafx_vector_math_tests tests/VectorMathTests.cpp)
    target_link_libraries(liberafx_vector_math_tests PRIVATE liberafx_runtime)
    add_test(NAME liberafx_vector_math_tests COMMAND liberafx_vector_math_tests)
    add_executable(liberafx_profile_recorder_tests tests/ProfileRecorderTests.cpp)

    target_link_libraries(liberafx_scene_binding_tests PRIVATE liberafx_runtime)
    add_test(NAME liberafx_scene_binding_tests COMMAND liberafx_scene_binding_tests)
    if(LIBERAFX_BUILD_INTEROP)
